          {success &&

              </button>
            ) : (
              <button className="primary" onClick={onFinish}>
                Перейти к карточкам →
              </button>
            ))}
        </div>
      </section>
    </main>
  );
}

// В вариантах ответа нельзя выделять только правильный факт: это превращается в подсказку.
// Поэтому даты, числа, сокращения и фамилии оформляются одинаково во ВСЕХ вариантах.
const optionFactSource = [
  "\\d{1,2}\\s+(?:января|февраля|марта|апреля|мая|июня|июля|августа|сентября|октября|ноября|декабря)\\s+\\d{4}\\s+года",
  "\\d{4}[–-]\\d{4}(?:\\s+годы)?",
  "\\d{4}(?:\\s+год(?:а|у|ом)?)?",
  "ВДНХ", "ВСХВ", "ВВЦ", "МГУ", "МИД", "ОВИО",
  "Циолковский", "Королёв", "Гагарин", "Мухина", "Иофан", "Мельчаков", "Щуко", "Столяров", "Топуридзе", "Константиновский", "Бажов",
  "\\d+",
].join("|");
const optionFactPattern = new RegExp(`(${optionFactSource})`, "g");
const optionFactExact = new RegExp(`^(?:${optionFactSource})$`);
function highlightOptionFacts(text: string) {
  return text.split(optionFactPattern).map((part, index) =>
    optionFactExact.test(part) ? <strong className="key-fact" key={`${part}-${index}`}>{part}</strong> : part,
  );
}

function mixHighriseQuestion(q: readonly [string, readonly string[], number]) {
  const variants = q[1].map((text, index) => ({ text, right: index === q[2] }));
  for (let i = variants.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [variants[i], variants[j]] = [variants[j], variants[i]];
  }
  return { title: q[0], options: variants.map(v => v.text), answer: variants.findIndex(v => v.right) };
}
